<script>

let trabajadores = JSON.parse(localStorage.getItem("trabajadores")) || [];
let editandoId = null;

/* 🔥 LISTA ÚNICA CORREGIDA DE CANELONES */
const zonasCanelones=[
"Ciudad de la Costa",
"Las Piedras",
"Las Toscas",
"El Pinar",
"Paso Carrasco",
"Canelones",
"La Floresta",
"Atlántida",

"Pando",

"Parque del Plata"
];

const oficios=[
"Electricista","Costura","Reposteria","Sanitario",
"Tecnico A.Acondicionado","Seguridad","Tecnico Electrodomesticos",
"Dentista","Veterinaria/o","Traslados","Cadeteria",
"Albañileria","Jardineria","Limpieza","Souvenirs,Regaleria"

function login(){
    if(document.getElementById("clave").value==="mery123"){
        document.getElementById("formulario").style.display="block";
        document.getElementById("adminPanel").style.display="none";
    }else{
        alert("Clave incorrecta");
    }
}

document.addEventListener("DOMContentLoaded", function(){
    cargarDepartamentos();
    cargarOficios();
    cargarFiltros();
});

function cargarDepartamentos(){
    let d=document.getElementById("departamento");

    d.innerHTML=
    <option value="">Seleccionar</option>
    <option value="Montevideo">Montevideo</option>
    <option value="Canelones">Canelones</option>;
}

function cargarZonas(){
    let d=document.getElementById("departamento")?.value;
    let z=document.getElementById("zona");

    z.innerHTML="<option value=''>Seleccionar</option>";

    let lista=[];

    if(d==="Montevideo"){
        lista=["Centro","Cordón","Pocitos","Buceo","Carrasco","Union","La Teja","Paso Molino","Capurro","Cerro","Maroñas","Prado","Aguada"];
    }

    if(d==="Canelones"){
        lista=zonasCanelones;
    }

    lista.forEach(x=>{
        z.innerHTML+=<option>${x}</option>;
    });
}

function cargarZonasFiltro(){
    let d=document.getElementById("filtroDepartamento")?.value;
    let z=document.getElementById("filtroZona");

    z.innerHTML="<option value=''>Localidad</option>";

    let lista=[];

    if(d==="Montevideo"){
        lista=["Centro","Cordón","Pocitos","Buceo","Carrasco","Union","La Teja","Paso Molino","Sayago","Cerro","Maroñas","Prado","Aguada"];
    }

    if(d==="Canelones"){
        lista=zonasCanelones;
    }

    lista.forEach(x=>{
        z.innerHTML+=<option>${x}</option>;
    });
}

function cargarOficios(){
    let o=document.getElementById("oficio");

    o.innerHTML="<option value=''>Seleccionar</option>";
    oficios.forEach(x=>{
        o.innerHTML+=<option>${x}</option>;
    });
}

function cargarFiltros(){
    let f=document.getElementById("filtroOficio");

    f.innerHTML="<option value=''>Oficio</option>";
    oficios.forEach(x=>{
        f.innerHTML+=<option>${x}</option>;
    });
}

function agregarTrabajador(){

    let nombre=document.getElementById("nombre").value;
    let tel=document.getElementById("telefono").value;

    if(!nombre || !tel){
        alert("Completá nombre y teléfono");
        return;
    }

    if(editandoId !== null){

        trabajadores = trabajadores.map(t=>{
            if(t.id === editandoId){
                t.nombre = nombre;
                t.tel = tel;
                t.departamento = document.getElementById("departamento").value;
                t.zona = document.getElementById("zona").value;
                t.oficio = document.getElementById("oficio").value;
            }
            return t;
        });

        editandoId = null;

    } else {

        trabajadores.push({
            id:Date.now(),
            nombre,
            tel,
            departamento:document.getElementById("departamento").value,
            zona:document.getElementById("zona").value,
            oficio:document.getElementById("oficio").value,
            calificacion:0,
            votos:0
        });
    }

    localStorage.setItem("trabajadores",JSON.stringify(trabajadores));

    document.getElementById("nombre").value="";
    document.getElementById("telefono").value="";
}

function whatsapp(t){
    window.open(https://wa.me/598${t});
}

function calificar(id,nota){
    trabajadores=trabajadores.map(t=>{
        if(t.id===id){
            let total=t.calificacion*t.votos;
            t.votos++;
            t.calificacion=(total+nota)/t.votos;
        }
        return t;
    });

    localStorage.setItem("trabajadores",JSON.stringify(trabajadores));
    buscar();
}

function editarTrabajador(id){

    let t = trabajadores.find(t => t.id === id);

    document.getElementById("nombre").value = t.nombre;
    document.getElementById("telefono").value = t.tel;
    document.getElementById("departamento").value = t.departamento;

    cargarZonas();

    setTimeout(()=>{
        document.getElementById("zona").value = t.zona;
    },50);

    document.getElementById("oficio").value = t.oficio;

    editandoId = id;
}

function eliminarTrabajador(id){

    trabajadores = trabajadores.filter(t => t.id !== id);

    localStorage.setItem("trabajadores", JSON.stringify(trabajadores));

    buscar();
}

function buscar(){

    let dep=document.getElementById("filtroDepartamento").value;
    let zona=document.getElementById("filtroZona").value;
    let oficio=document.getElementById("filtroOficio").value;

    let r=trabajadores.filter(t=>
        (dep===""||t.departamento===dep)&&
        (zona===""||t.zona===zona)&&
        (oficio===""||t.oficio===oficio)
    );

    let c=document.getElementById("resultados");
    c.innerHTML="";

    if(r.length===0){
        c.innerHTML="No hay resultados";
        return;
    }

    r.forEach(t=>{
        c.innerHTML+=
        <div class="card">
        <b>${t.nombre}</b><br>
        📍 ${t.departamento} - ${t.zona}<br>
        🛠 ${t.oficio}<br>
        📱 ${t.tel}<br>
        ⭐ ${t.calificacion.toFixed(1)} (${t.votos})<br>

        <button onclick="whatsapp('${t.tel}')">WhatsApp</button>
        <button onclick="calificar(${t.id},5)">⭐5</button>
        <button onclick="editarTrabajador(${t.id})">✏️ Editar</button>
        <button onclick="eliminarTrabajador(${t.id})">🗑️ Eliminar</button>
        </div>;
    });
}
